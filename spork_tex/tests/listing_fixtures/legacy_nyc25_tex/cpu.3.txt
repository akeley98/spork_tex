@proc
def nyc25_gemm_cpu(M: size, N: size, K: size,
                   A: f32[M, K] @ DRAM,
                   B: f32[K, N] @ DRAM,
                   C: f32[M, N] @ DRAM):
  for m in seq(0, M):
    for n in seq(0, N):
      accum: f32 @ DRAM
      accum = 0
      for k in seq(0, K):
        accum += A[m, k] * B[k, n]
      C[m, n] = accum
